using System;

class Program
{
    static void Main(string[] args)
    {
        Console.WriteLine("TUGAS MUHAMMAD HAFIDZ TAIMIYAH SETYAJI");
        Console.WriteLine("KELAS 10 RPL 1");
        Console.WriteLine("REMEDIAL DPK PAK RIAN");
        Console.WriteLine("PEMINJAMAN BUKU");
        Console.Write("Masukkan jumlah data: ");

        int jumlahData = int.Parse(Console.ReadLine() ?? "0");

        string[] nama = new string[jumlahData];
        string[] genre = new string[jumlahData];
        string[,] dataPeminjaman = new string[jumlahData, 4];
        DateTime[] tanggalPinjam = new DateTime[jumlahData];
        DateTime[] tanggalKembali = new DateTime[jumlahData];

        for (int i = 0; i < jumlahData; i++)
        {
            Console.WriteLine($"\n{i + 1}. Masukkan jumlah data peminjam");

            Console.Write("Nama peminjam: ");
            nama[i] = Console.ReadLine();

            Console.Write("ID buku: ");
            dataPeminjaman[i, 0] = Console.ReadLine();

            Console.Write("Judul buku: ");
            dataPeminjaman[i, 1] = Console.ReadLine();

            Console.Write("Genre buku: ");
            genre[i] = Console.ReadLine();

            Console.Write("Tanggal pinjam (contoh: 2026-09-29): ");
            tanggalPinjam[i] = DateTime.Parse(Console.ReadLine());

            Console.Write("Tanggal kembali (contoh: 2026-09-29): ");
            tanggalKembali[i] = DateTime.Parse(Console.ReadLine());

            dataPeminjaman[i, 2] = tanggalPinjam[i].ToString("yyyy-MM-dd");
            dataPeminjaman[i, 3] = tanggalKembali[i].ToString("yyyy-MM-dd");
        }

        Console.WriteLine("\nDAFTAR PEMINJAMAN BUKU");
        Console.WriteLine("Peminjam | ID | Judul buku | Genre | Tgl pinjam | Tgl kembali");
        for (int i = 0; i < jumlahData; i++)
        {
            Console.WriteLine(
                $"{i + 1}. {nama[i]} | {dataPeminjaman[i, 0]} | {dataPeminjaman[i, 1]} | {genre[i]} | {dataPeminjaman[i, 2]} | {dataPeminjaman[i, 3]}");
        }

        Console.WriteLine("\nLama peminjaman:");
        for (int i = 0; i < jumlahData; i++)
        {
            int lamaPinjam = (tanggalKembali[i] - tanggalPinjam[i]).Days;
            Console.WriteLine(
                $"{nama[i]}: {dataPeminjaman[i, 2]} -> {dataPeminjaman[i, 3]} : {lamaPinjam} hari");
        }
    }
}
